Projeto de modelagem de banco de dados

Sistema de Gestão Operacional e Vendas — Kalunga S.A.

Projeto acadêmico, sem vínculo com a empresa. A Kalunga S.A. é usada apenas como estudo de caso de domínio. Este não é um sistema real nem documentação interna da empresa: requisitos, regras, estrutura e dados são hipotéticos, elaborados para a disciplina SQL (ADS/FSA), sem participação ou autorização da Kalunga.

Escopo

Protótipo acadêmico de banco de dados relacional para uma unidade de varejo de papelaria simulada, tendo a Kalunga S.A. como referência de domínio. O banco centraliza cadastros de clientes, fornecedores, categorias, produtos e usuários; registra vendas presenciais com itens, desconto e um pagamento integral; controla entradas manuais e saídas automáticas de estoque; oferece consultas de saldo e relatórios de vendas; e mantém trilha de auditoria das alterações sensíveis. Ficam fora desta versão: comércio eletrônico, emissão fiscal (NF-e/NFC-e), integrações com ERP, compras, devoluções e estornos.

Cultura de dados (Data Literacy)

Este dicionário traduz o modelo físico em linguagem de negócio: cada coluna informa para que serve, de onde vem, para onde vai, quem pode acessá-la e como é classificada pela LGPD. Segundo o DAMA-DMBOK, metadados documentados são a base da governança, porque permitem que atendentes, gerentes, estoquistas e administradores interpretem os dados da mesma forma. No COBIT, o dicionário apoia a gestão de dados como ativo (APO14), dando rastreabilidade entre requisitos, regras de negócio e controles técnicos. Para a unidade de varejo, isso significa menos divergência de estoque, vendas auditáveis e tratamento de dados pessoais com finalidade e minimização.

Privacidade (LGPD / GDPR)

  • [DPS] Dado pessoal sensível (Lei nº 13.709/2018, art. 5º, II). Nenhuma coluna deste modelo armazena dado sensível.
  • [DP] Dado pessoal: identifica ou pode identificar uma pessoa natural, direta ou indiretamente.
  • [N/A] Não aplicável: dado operacional, sem vínculo com pessoa natural.

Controle de acesso (RBAC)

  • C cria · L lê · A atualiza · I inativa
  • Perfis: ATENDENTE, GERENTE, ESTOQUISTA e ADMINISTRADOR (RN11).
  • Exclusão física não é concedida a nenhum perfil; registros com histórico são inativados (PR02).

DROP DATABASE IF EXISTS kalunga_vendas;
CREATE DATABASE kalunga_vendas
    DEFAULT CHARACTER SET utf8mb4
    DEFAULT COLLATE utf8mb4_0900_ai_ci;
USE kalunga_vendas;

-- =====================================================================
-- BLOCO 1: CADASTROS E ACESSOS
-- =====================================================================

-- RF01 / RN01 / PR01 - Clientes (CPF opcional e unico quando informado)
CREATE TABLE cliente (
    id_cliente  INT          NOT NULL AUTO_INCREMENT,
    nome        VARCHAR(120) NOT NULL,
    cpf         CHAR(11)     NULL,
    telefone    VARCHAR(20)  NULL,
    email       VARCHAR(150) NULL,
    ativo       BOOLEAN      NOT NULL DEFAULT 1,
    CONSTRAINT pk_cliente PRIMARY KEY (id_cliente),
    CONSTRAINT uk_cliente_cpf UNIQUE (cpf),
    CONSTRAINT ck_cliente_cpf CHECK (cpf IS NULL OR cpf REGEXP '^[0-9]{11}$')
) ENGINE = InnoDB;

-- RF11 / RN11 / RNF01 - Usuarios do sistema e perfis de acesso
CREATE TABLE usuario (
    id_usuario  INT          NOT NULL AUTO_INCREMENT,
    nome        VARCHAR(120) NOT NULL,
    login       VARCHAR(80)  NOT NULL,
    senha_hash  VARCHAR(255) NOT NULL,
    perfil      ENUM('ATENDENTE', 'GERENTE', 'ESTOQUISTA', 'ADMINISTRADOR') NOT NULL,
    ativo       BOOLEAN      NOT NULL DEFAULT 1,
    CONSTRAINT pk_usuario PRIMARY KEY (id_usuario),
    CONSTRAINT uk_usuario_login UNIQUE (login)
) ENGINE = InnoDB;

-- =====================================================================
-- BLOCO 2: CATALOGO E ESTOQUE
-- =====================================================================

-- RF02 / RN02 - Fornecedores
CREATE TABLE fornecedor (
    id_fornecedor INT          NOT NULL AUTO_INCREMENT,
    razao_social  VARCHAR(150) NOT NULL,
    cnpj          CHAR(14)     NOT NULL,
    telefone      VARCHAR(20)  NULL,
    email         VARCHAR(150) NULL,
    CONSTRAINT pk_fornecedor PRIMARY KEY (id_fornecedor),
    CONSTRAINT uk_fornecedor_cnpj UNIQUE (cnpj),
    CONSTRAINT ck_fornecedor_cnpj CHECK (cnpj REGEXP '^[0-9]{14}$')
) ENGINE = InnoDB;

-- RF03 / RN03 - Categorias de produtos
CREATE TABLE categoria (
    id_categoria INT          NOT NULL AUTO_INCREMENT,
    nome         VARCHAR(100) NOT NULL,
    descricao    VARCHAR(255) NULL,
    CONSTRAINT pk_categoria PRIMARY KEY (id_categoria),
    CONSTRAINT uk_categoria_nome UNIQUE (nome)
) ENGINE = InnoDB;

-- RF04 / RN03 / RN04 - Produtos (saldo inicia em zero; carga via RF08)
CREATE TABLE produto (
    id_produto     INT           NOT NULL AUTO_INCREMENT,
    id_categoria   INT           NOT NULL,
    id_fornecedor  INT           NOT NULL,
    descricao      VARCHAR(200)  NOT NULL,
    preco          DECIMAL(10,2) NOT NULL,
    saldo_estoque  INT           NOT NULL DEFAULT 0,
    estoque_minimo INT           NOT NULL DEFAULT 0,
    ativo          BOOLEAN       NOT NULL DEFAULT 1,
    CONSTRAINT pk_produto PRIMARY KEY (id_produto),
    CONSTRAINT fk_produto_categoria FOREIGN KEY (id_categoria)
        REFERENCES categoria (id_categoria) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT fk_produto_fornecedor FOREIGN KEY (id_fornecedor)
        REFERENCES fornecedor (id_fornecedor) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT ck_produto_preco CHECK (preco >= 0),
    CONSTRAINT ck_produto_saldo CHECK (saldo_estoque >= 0),
    CONSTRAINT ck_produto_estoque_minimo CHECK (estoque_minimo >= 0)
) ENGINE = InnoDB;

-- =====================================================================
-- BLOCO 3: VENDAS E TRANSACIONAL
-- =====================================================================

-- RF05 / RF06 / RN05 / RN08 - Vendas (EM_ABERTO -> FINALIZADA)
CREATE TABLE venda (
    id_venda    INT           NOT NULL AUTO_INCREMENT,
    id_cliente  INT           NULL,
    id_usuario  INT           NOT NULL,
    data_venda  DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    situacao    ENUM('EM_ABERTO', 'FINALIZADA') NOT NULL DEFAULT 'EM_ABERTO',
    desconto    DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    valor_total DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    CONSTRAINT pk_venda PRIMARY KEY (id_venda),
    CONSTRAINT fk_venda_cliente FOREIGN KEY (id_cliente)
        REFERENCES cliente (id_cliente) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT fk_venda_usuario FOREIGN KEY (id_usuario)
        REFERENCES usuario (id_usuario) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT ck_venda_desconto CHECK (desconto >= 0),
    CONSTRAINT ck_venda_valor_total CHECK (valor_total >= 0)
) ENGINE = InnoDB;

-- RF05 / RN06 / RN07 - Itens da venda (preco historico congelado)
CREATE TABLE item_venda (
    id_item_venda  INT           NOT NULL AUTO_INCREMENT,
    id_venda       INT           NOT NULL,
    id_produto     INT           NOT NULL,
    quantidade     INT           NOT NULL,
    preco_unitario DECIMAL(10,2) NOT NULL,
    CONSTRAINT pk_item_venda PRIMARY KEY (id_item_venda),
    CONSTRAINT uk_item_venda_venda_produto UNIQUE (id_venda, id_produto),
    CONSTRAINT fk_item_venda_venda FOREIGN KEY (id_venda)
        REFERENCES venda (id_venda) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT fk_item_venda_produto FOREIGN KEY (id_produto)
        REFERENCES produto (id_produto) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT ck_item_venda_quantidade CHECK (quantidade > 0),
    CONSTRAINT ck_item_venda_preco CHECK (preco_unitario > 0)
) ENGINE = InnoDB;

-- RF07 / RN09 - Pagamento integral (zero ou um por venda)
CREATE TABLE pagamento (
    id_pagamento   INT           NOT NULL AUTO_INCREMENT,
    id_venda       INT           NOT NULL,
    forma          VARCHAR(30)   NOT NULL,
    valor          DECIMAL(10,2) NOT NULL,
    data_pagamento DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_pagamento PRIMARY KEY (id_pagamento),
    CONSTRAINT uk_pagamento_venda UNIQUE (id_venda),
    CONSTRAINT fk_pagamento_venda FOREIGN KEY (id_venda)
        REFERENCES venda (id_venda) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT ck_pagamento_valor CHECK (valor >= 0)
) ENGINE = InnoDB;

-- RF08 / RN10 - Movimentacoes de estoque (ENTRADA manual / SAIDA por venda)
CREATE TABLE movimentacao_estoque (
    id_movimentacao   INT          NOT NULL AUTO_INCREMENT,
    id_produto        INT          NOT NULL,
    id_usuario        INT          NOT NULL,
    id_venda          INT          NULL,
    tipo              ENUM('ENTRADA', 'SAIDA') NOT NULL,
    quantidade        INT          NOT NULL,
    data_movimentacao DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    motivo            VARCHAR(255) NULL,
    CONSTRAINT pk_movimentacao_estoque PRIMARY KEY (id_movimentacao),
    CONSTRAINT uk_movimentacao_venda_produto_tipo UNIQUE (id_venda, id_produto, tipo),
    CONSTRAINT fk_movimentacao_produto FOREIGN KEY (id_produto)
        REFERENCES produto (id_produto) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT fk_movimentacao_usuario FOREIGN KEY (id_usuario)
        REFERENCES usuario (id_usuario) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT fk_movimentacao_venda FOREIGN KEY (id_venda)
        REFERENCES venda (id_venda) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT ck_movimentacao_quantidade CHECK (quantidade > 0),
    CONSTRAINT ck_movimentacao_tipo_venda CHECK (
        (tipo = 'ENTRADA' AND id_venda IS NULL) OR
        (tipo = 'SAIDA'   AND id_venda IS NOT NULL)
    )
) ENGINE = InnoDB;

-- =====================================================================
-- BLOCO 4: AUDITORIA
-- =====================================================================

-- RF12 / RN12 / PR01 - Trilha de auditoria append-only
CREATE TABLE log_auditoria (
    id_log_auditoria INT         NOT NULL AUTO_INCREMENT,
    id_usuario       INT         NOT NULL,
    nome_tabela      VARCHAR(80) NOT NULL,
    id_registro      INT         NOT NULL,
    operacao         VARCHAR(20) NOT NULL,
    data_evento      DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    valor_anterior   JSON        NULL,
    valor_novo       JSON        NULL,
    CONSTRAINT pk_log_auditoria PRIMARY KEY (id_log_auditoria),
    CONSTRAINT fk_log_auditoria_usuario FOREIGN KEY (id_usuario)
        REFERENCES usuario (id_usuario) ON DELETE RESTRICT ON UPDATE RESTRICT,
    CONSTRAINT ck_log_auditoria_operacao CHECK (operacao IN ('INSERT', 'UPDATE', 'INATIVACAO'))
) ENGINE = InnoDB;

-- =====================================================================
-- INDICES DE DESEMPENHO (secao 6.3 / RNF03)

CREATE INDEX idx_venda_data_venda ON venda (data_venda);

CREATE TRIGGER trg_produto_auditoria_update
AFTER UPDATE ON produto
FOR EACH ROW
BEGIN
    IF NOT (OLD.descricao <=> NEW.descricao AND OLD.preco <=> NEW.preco
            AND OLD.estoque_minimo <=> NEW.estoque_minimo AND OLD.ativo <=> NEW.ativo
            AND OLD.id_categoria <=> NEW.id_categoria
            AND OLD.id_fornecedor <=> NEW.id_fornecedor) THEN
        INSERT INTO log_auditoria (id_usuario, nome_tabela, id_registro, operacao, valor_anterior, valor_novo)
        VALUES (@id_usuario_sessao, 'produto', NEW.id_produto,
                IF(OLD.ativo = 1 AND NEW.ativo = 0, 'INATIVACAO', 'UPDATE'),
                JSON_OBJECT('descricao', OLD.descricao, 'preco', OLD.preco,
                            'estoque_minimo', OLD.estoque_minimo, 'ativo', OLD.ativo),
                JSON_OBJECT('descricao', NEW.descricao, 'preco', NEW.preco,
                            'estoque_minimo', NEW.estoque_minimo, 'ativo', NEW.ativo));
    END IF;
END$$

BEFORE DELETE ON movimentacao_estoque
FOR EACH ROW
BEGIN
    SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'RN10: movimentacoes de estoque nao podem ser excluidas.';
END$$

-- RF08 / RN04 / RN10 - Entrada manual de estoque (transacional)
CREATE PROCEDURE sp_registrar_entrada_estoque (
    IN p_id_produto INT,
    IN p_quantidade INT,
    IN p_motivo     VARCHAR(255),
    IN p_id_usuario INT
)
BEGIN
    DECLARE v_existe INT DEFAULT 0;

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    IF p_quantidade IS NULL OR p_quantidade <= 0 THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'RF08: a quantidade de entrada deve ser positiva.';
    END IF;

    START TRANSACTION;

    SELECT COUNT(*) INTO v_existe
      FROM produto
     WHERE id_produto = p_id_produto
       FOR UPDATE;

    IF v_existe = 0 THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'RF08: produto inexistente.';
    END IF;

    INSERT INTO movimentacao_estoque (id_produto, id_usuario, id_venda, tipo, quantidade, motivo)
    VALUES (p_id_produto, p_id_usuario, NULL, 'ENTRADA', p_quantidade, p_motivo);

    UPDATE produto
       SET saldo_estoque = saldo_estoque + p_quantidade
     WHERE id_produto = p_id_produto;

    COMMIT;
END$$

-- RF05 / RF06 / RF07 / RN05 / RN06 / RN08 / RN09 / RN10 / RNF05 / RNF08
-- Finalizacao atomica da venda: valida itens, cliente, saldo, total e
-- pagamento; baixa o estoque; grava as saidas e muda a situacao.
CREATE PROCEDURE sp_finalizar_venda (
    IN p_id_venda INT
)
BEGIN
    DECLARE v_situacao   VARCHAR(20);
    DECLARE v_id_cliente INT;
    DECLARE v_id_usuario INT;
    DECLARE v_desconto   DECIMAL(10,2);
    DECLARE v_subtotal   DECIMAL(10,2);
    DECLARE v_total      DECIMAL(10,2);
    DECLARE v_qtd_itens  INT DEFAULT 0;
    DECLARE v_qtd_pagto  INT DEFAULT 0;
    DECLARE v_valor_pagto DECIMAL(10,2);
    DECLARE v_invalidos  INT DEFAULT 0;
    DECLARE v_bloqueados INT DEFAULT 0;

    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SET @em_finalizacao = NULL;
        RESIGNAL;
    END;

    START TRANSACTION;

    -- 1. Trava a venda
    SELECT situacao, id_cliente, id_usuario, desconto
      INTO v_situacao, v_id_cliente, v_id_usuario, v_desconto
      FROM venda
     WHERE id_venda = p_id_venda
       FOR UPDATE;

    IF v_situacao IS NULL THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'RN05: venda inexistente.';
    END IF;
    IF v_situacao <> 'EM_ABERTO' THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'RN05: venda ja FINALIZADA.';
    END IF;

    -- 2. Cliente ativo quando informado
    IF v_id_cliente IS NOT NULL
       AND (SELECT ativo FROM cliente WHERE id_cliente = v_id_cliente) = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'RN05: cliente inativo.';
    END IF;

    -- 3. Ao menos um item
    SELECT COUNT(*) INTO v_qtd_itens FROM item_venda WHERE id_venda = p_id_venda;
    IF v_qtd_itens = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'RN05: venda sem itens.';
    END IF;

    -- 4. Trava os produtos da venda (RNF08): a leitura percorre a chave
    --    primaria em ordem crescente de id_produto, evitando deadlock
    SELECT COUNT(*) INTO v_bloqueados
      FROM produto
     WHERE id_produto IN (SELECT id_produto FROM item_venda WHERE id_venda = p_id_venda)
       FOR UPDATE;

    -- 5. Produtos ativos, preco positivo e saldo suficiente (RN04 / RN06)
    SELECT COUNT(*) INTO v_invalidos
      FROM item_venda i
      JOIN produto p ON p.id_produto = i.id_produto
     WHERE i.id_venda = p_id_venda
       AND (p.ativo = 0 OR i.preco_unitario <= 0 OR i.quantidade > p.saldo_estoque);

    IF v_invalidos > 0 THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'RN04/RN06: produto inativo, preco invalido ou saldo insuficiente.';
    END IF;

    -- 6. Total = soma(quantidade x preco_unitario) - desconto (RN08)
    SELECT SUM(quantidade * preco_unitario) INTO v_subtotal
      FROM item_venda
     WHERE id_venda = p_id_venda;

    IF v_desconto > v_subtotal THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'RN08: desconto maior que o subtotal.';
    END IF;
    SET v_total = v_subtotal - v_desconto;

    -- 7. Exatamente um pagamento integral (RN09)
    SELECT COUNT(*), MAX(valor) INTO v_qtd_pagto, v_valor_pagto
      FROM pagamento
     WHERE id_venda = p_id_venda;

    IF v_qtd_pagto <> 1 OR v_valor_pagto <> v_total THEN
        SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT = 'RN09: pagamento ausente ou diferente do total.';
    END IF;

    -- 8. Baixa de estoque e saida unica por produto (RN06 / RN10)
    UPDATE produto p
      JOIN item_venda i ON i.id_produto = p.id_produto
       SET p.saldo_estoque = p.saldo_estoque - i.quantidade
     WHERE i.id_venda = p_id_venda;

    INSERT INTO movimentacao_estoque (id_produto, id_usuario, id_venda, tipo, quantidade, motivo)
    SELECT i.id_produto, v_id_usuario, p_id_venda, 'SAIDA', i.quantidade,
           CONCAT('Venda ', p_id_venda)
      FROM item_venda i
     WHERE i.id_venda = p_id_venda;

    -- 9. Muda a situacao e grava o total
    SET @em_finalizacao = 1;
    UPDATE venda
       SET situacao = 'FINALIZADA',
           valor_total = v_total
     WHERE id_venda = p_id_venda;
    SET @em_finalizacao = NULL;

    COMMIT;
END$$

DELIMITER ;

-- =====================================================================
-- VISAO DE APOIO (RF09 - produtos com estoque baixo)
-- =====================================================================

CREATE VIEW vw_estoque_baixo AS
SELECT p.id_produto,
       p.descricao,
       c.nome AS categoria,
       p.saldo_estoque,
       p.estoque_minimo
  FROM produto p
  JOIN categoria c ON c.id_categoria = p.id_categoria
 WHERE p.ativo = 1
   AND p.saldo_estoque < p.estoque_minimo;
